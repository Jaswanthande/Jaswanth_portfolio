# Scikit-learn, NLTK, TF-IDF, Cosine Similarity
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import cosine_similarity
import re, json, nltk
class ResumeScreeningEngine:
def __init__(self, target_jd):
self.target_jd = target_jd
self.vectorizer = TfidfVectorizer(stop_words='english')
def evaluate_candidate(self, resume_text):
# Tokenize & compute TF-IDF vector similarity
matrix = self.vectorizer.fit_transform([self.target_jd, resume_text])
score = cosine_similarity(matrix[0:1], matrix[1:2])[0][0]
return { "status": "QUALIFIED", "match": f"{score * 100:.1f}%" }
